// Engine stubs isolate dispatch and callback completion. They do not simulate spawning.
class Program
{
    static void Check(bool condition, string name)
    {
        if (!condition) throw new Exception(name);
    }
    static int Main()
    {
        try { Run(); return 0; }
        catch (Exception error) { Console.Error.WriteLine(error.Message); return 1; }
    }
    static void Run()
    {
        foreach (bool dlc in new[] { false, true })
        foreach (bool playerSide in new[] { false, true })
        foreach (string place in new[] { "battle", "location", "naval" })
        {
            CampaignHelpers.Enabled = dlc;
            Mission.Current = new() { IsNavalBattle = place == "naval" };
            CampaignMission.Current = new() { Location = place == "location" ? new object() : null };
            var settings = new Settings { OnPlayerSide = playerSide };
            var replies = new List<string>();
            new SummonHero().ExecuteInternal(new(), new(), settings, replies.Add, error => throw new Exception(error));
            Check(replies.Count == (place == "location" ? 1 : 0), "Battle spawning must wait for the mission tick");
            BLTSummonBehavior.Current.Tick();
            string expected = dlc && place == "naval" ? "naval" : place == "location" ? "location" : "battle";
            Check(replies.SequenceEqual(new[] { expected }), $"Missing/wrong summon dispatch: DLC={dlc}, playerSide={playerSide}, mission={place}");
        }
        Mission.Current = null;
        int failures = 0;
        new SummonHero().ExecuteInternal(new(), new(), new Settings(), _ => throw new Exception("No mission succeeded"), _ => failures++);
        Check(failures == 1, "No mission must return one failure, without throwing");
        Mission.Current = new();
        CampaignMission.Current = null;
        var fallback = new List<string>();
        new SummonHero().ExecuteInternal(new(), new(), new Settings(), fallback.Add, _ => throw new Exception("Null campaign mission failed"));
        BLTSummonBehavior.Current.Tick();
        Check(fallback.SequenceEqual(new[] { "battle" }), "A battle without a campaign location must dispatch");
        Console.WriteLine("PASS actual summon dispatch: DLC on/off, summon/attack sides, land/location/naval routing, mission-tick deferral, one terminal callback, no-mission rejection, null campaign location.");
    }
}

class BLTSummonBehavior
{
    public static BLTSummonBehavior Current = new();
    readonly Queue<Action> queue = new();
    public void DoNextTick(Action action) => queue.Enqueue(action);
    public void Tick() { int count = queue.Count; while (count-- > 0) queue.Dequeue()(); }
}
