    public static class Extensions
    {
        public static T SelectRandom<T>(this IEnumerable<T> source) => source.FirstOrDefault();
        public static string Translate(this string text, params (string, object)[] args) => text;
        public static bool HasAnyFlag(this ItemFlags flags, ItemFlags other) => (flags & other) != 0;
        public static bool IsEquipmentType(this ItemObject item, EquipmentType type) => (type == EquipmentType.OneHandedSword && item.ItemType == ItemObject.ItemTypeEnum.OneHandedWeapon && !item.StringId.Contains("mace")) || (type == EquipmentType.Bow && item.ItemType == ItemObject.ItemTypeEnum.Bow) || (type == EquipmentType.Arrows && item.ItemType == ItemObject.ItemTypeEnum.Arrows) || (type == EquipmentType.OneHandedMace && item.StringId.Contains("mace")) || (type == EquipmentType.ThrowingAxes && item.ItemType == ItemObject.ItemTypeEnum.Thrown) || (type == EquipmentType.Shield && item.ItemType == ItemObject.ItemTypeEnum.Shield);
        public static IEnumerable<(EquipmentElement element, EquipmentIndex index)> YieldEquipmentSlots(this Equipment e)
            => Enumerable.Range(0, 11).Select(i => (e[(EquipmentIndex)i], (EquipmentIndex)i));
        public static IEnumerable<(EquipmentElement element, EquipmentIndex index)> YieldFilledEquipmentSlots(this Equipment e) => e.YieldEquipmentSlots().Where(x => !x.element.IsEmpty);
        public static IEnumerable<(EquipmentElement element, EquipmentIndex index)> YieldWeaponSlots(this Equipment e) => e.YieldEquipmentSlots().Take(4);
        public static IEnumerable<(EquipmentElement element, EquipmentIndex index)> YieldFilledWeaponSlots(this Equipment e) => e.YieldWeaponSlots().Where(x => !x.element.IsEmpty);
    }

public static class Program
{
    static int checks;
    static void Check(bool value, string description) { if (!value) throw new Exception(description); Console.WriteLine("PASS " + description); checks++; }
    static ItemObject Item(string id, int tier, ItemObject.ItemTypeEnum type = ItemObject.ItemTypeEnum.BodyArmor) => new() { StringId = id, Tier = tier, ItemType = type };
    static void Reset()
    {
        Campaign.Current = new(); CharacterObject.All.Clear(); CampaignHelpers.AllItems.Clear();
        HarmonyLib.AccessTools.Installed = true; TaomEquipmentCompatibility.Reset();
        BLTAdoptAHeroModule.CommonConfig.RestrictedItemIds.Clear();
    }
    static void Donor(int race, params ItemObject[] items)
    {
        var c = new CharacterObject { Race = race };
        foreach (var item in items) { var e = new Equipment(); e[EquipmentIndex.Body] = new(item); c.BattleEquipments.Add(e); }
        CharacterObject.All.Add(c); CampaignHelpers.AllItems.AddRange(items);
    }
    static Hero Hero(int race, ItemObject initial = null)
    {
        var h = new Hero { CharacterObject = new() { Race = race, IsHero = true } };
        h.BattleEquipment[EquipmentIndex.Body] = new(initial); h.CivilianEquipment[EquipmentIndex.Body] = new(initial); return h;
    }
    public static void Main()
    {
        Reset();
        var human = Item("human", 0); var east = Item("east", 0); var elf = Item("elf", 0); var elite = Item("elf-elite", 5);
        var sword = Item("elf-sword", 0, ItemObject.ItemTypeEnum.OneHandedWeapon);
        Donor(0, human); Donor(1, east); Donor(2, elf, elite, sword);
        Check(!EquipHero.CanUseItem(Hero(2), human, true, false), "Elf excludes human gear even when ability is ignored");
        Check(!EquipHero.CanUseItem(Hero(1), elf, true, false), "Custom human race does not receive elf armour");
        foreach (var race in new[] { 1, 2 })
        {
            var h = Hero(race, elite); h.CustomItems.Add(new(elite));
            EquipHero.Adopt(h, new Settings(), h.Class);
            Check(h.Tier == 0, $"Race {race} starts at displayed Tier 1");
            Check(h.BattleEquipment.YieldFilledEquipmentSlots().All(x => x.element.Item.Tier <= 0), $"Race {race} has no template/inherited Tier 6 battle gear");
            Check(h.CivilianEquipment.YieldFilledEquipmentSlots().All(x => x.element.Item.Tier <= 0), $"Race {race} has no Tier 6 civilian gear");
            Check(h.BattleEquipment[EquipmentIndex.Body].Item == (race == 1 ? east : elf), $"Race {race} receives its own compatible armour");
        }
        var noGear = Hero(2, elite); EquipHero.Adopt(noGear, new Settings { StartingEquipmentTier = 0 }, noGear.Class);
        Check(noGear.Tier == -1 && !noGear.BattleEquipment.YieldFilledEquipmentSlots().Any() && !noGear.CivilianEquipment.YieldFilledEquipmentSlots().Any(), "Starting tier zero clears both loadouts for custom races");
        Reset(); Donor(2, elite);
        var missing = Hero(2, elite); EquipHero.Adopt(missing, new Settings(), missing.Class);
        Check(!missing.BattleEquipment.YieldFilledEquipmentSlots().Any(), "No low-tier kit means an empty slot, never elite fallback");
        var changed = true; EquipHero.Paid(missing, new Settings(), _ => changed = true, _ => changed = false);
        Check(!changed && missing.Gold == 1000 && missing.Tier == 0, "No suitable equipment does not charge or advance tier");
        Reset(); var culture = new CultureObject(); elf.Culture = culture; elite.Culture = culture; Donor(2, elf, elite);
        Check(EquipHero.FindRandomTieredEquipment(0, Hero(2), false, cultureFilter: culture, cultureFilterSpecified: true) == elf, "Culture selection obeys tier instead of taking the best item");
        var foreign = new CultureObject(); human.Culture = foreign; Donor(0, human); TaomEquipmentCompatibility.Reset();
        var fallback = Hero(2); EquipHero.UpgradeEquipment(fallback, 0, fallback.Class, true, foreign, true);
        Check(fallback.BattleEquipment[EquipmentIndex.Body].Item == elf, "Culture fallback retains race and tier checks");
        var upgraded = Item("elf-tier2", 1); Donor(2, upgraded); TaomEquipmentCompatibility.Reset();
        EquipHero.Paid(fallback, new Settings(), _ => changed = true, _ => changed = false);
        Check(changed && fallback.BattleEquipment[EquipmentIndex.Body].Item == upgraded && fallback.Gold == 900 && fallback.Tier == 1, "Non-human upgrade applies gear and charges once");
        var alternative = Item("elf-tier2-alternate", 1); Donor(2, alternative); TaomEquipmentCompatibility.Reset();
        EquipHero.Paid(fallback, new Settings { ReequipInsteadOfUpgrade = true }, _ => changed = true, _ => changed = false);
        Check(changed && fallback.BattleEquipment[EquipmentIndex.Body].Item == alternative && fallback.Tier == 1, "Non-human re-equip selects an alternative at the same tier");
        Reset();
        var home = new CultureObject(); var other = new CultureObject();
        var homeT2 = Item("home-tier2", 1, ItemObject.ItemTypeEnum.OneHandedWeapon); homeT2.Culture = home;
        var foreignT1 = Item("foreign-tier1", 0, ItemObject.ItemTypeEnum.OneHandedWeapon); foreignT1.Culture = other;
        var homeT6 = Item("home-tier6", 5, ItemObject.ItemTypeEnum.OneHandedWeapon); homeT6.Culture = home;
        Donor(1, homeT2, foreignT1, homeT6);
        var starter = Hero(1); starter.Culture = home;
        EquipHero.Adopt(starter, new Settings(), starter.Class);
        Check(starter.Tier == 0 && starter.BattleEquipment[EquipmentIndex.WeaponItemBeginSlot].Item == homeT2, "Same-culture Tier 2 fills Tier 1 class slot without raising viewer tier");
        CampaignHelpers.AllItems.Remove(homeT2);
        Check(EquipHero.FindRandomTieredEquipment(0, starter, false) == foreignT1, "Large same-culture gap falls back to compatible foreign Tier 1");
        CampaignHelpers.AllItems.Add(homeT2);
        Check(EquipHero.FindRandomTieredEquipment(2, starter, false) == homeT2, "Same-culture lower tier preferred within one tier");
        Check(EquipHero.FindRandomTieredEquipment(0, starter, false, EquipHero.FindFlags.RequireExactTier) == foreignT1, "Exact tier flag stays exact across cultures");
        BLTAdoptAHeroModule.CommonConfig.RestrictedItemIds.Add(homeT2.StringId);
        Check(EquipHero.FindRandomTieredEquipment(0, starter, false) == foreignT1, "Fallback cannot bypass restricted items");
        CampaignHelpers.AllItems.Remove(foreignT1);
        var trollSword = Item("troll-sword", 0, ItemObject.ItemTypeEnum.OneHandedWeapon); trollSword.Culture = other;
        Donor(8, trollSword); TaomEquipmentCompatibility.Reset();
        Check(EquipHero.FindRandomTieredEquipment(0, starter, false) == null, "Fallback rejects troll-only sword and distant Tier 6 gear");
        Reset();
        var mace = Item("mace", 0, ItemObject.ItemTypeEnum.OneHandedWeapon); Donor(1, mace);
        var meleeStarter = Hero(1); EquipHero.Adopt(meleeStarter, new Settings(), meleeStarter.Class);
        Check(meleeStarter.BattleEquipment[EquipmentIndex.Weapon0].Item == mace && meleeStarter.Tier == 0, "Last resort gives a Tier 1 mace for a missing sword");
        mace.Tier = 1;
        EquipHero.Adopt(meleeStarter, new Settings(), meleeStarter.Class);
        Check(meleeStarter.BattleEquipment[EquipmentIndex.Weapon0].IsEmpty, "Broad weapon fallback stays at exact tier");
        Reset();
        var crossbow = Item("crossbow", 0, ItemObject.ItemTypeEnum.Crossbow);
        var bolts = Item("bolts", 0, ItemObject.ItemTypeEnum.Bolts);
        Donor(1, crossbow, bolts);
        var archer = Hero(1); archer.Class.SlotItems = new[] { EquipmentType.Bow, EquipmentType.Arrows };
        EquipHero.Adopt(archer, new Settings(), archer.Class);
        Check(archer.BattleEquipment[EquipmentIndex.Weapon0].Item == crossbow && archer.BattleEquipment[EquipmentIndex.Weapon1].Item == bolts, "Missing bow gets crossbow with matching bolts");
        CampaignHelpers.AllItems.Remove(bolts);
        EquipHero.Adopt(archer, new Settings(), archer.Class);
        Check(archer.BattleEquipment[EquipmentIndex.Weapon0].IsEmpty, "No unusable ranged fallback without matching ammunition");
        var thrown = Item("javelin", 0, ItemObject.ItemTypeEnum.Thrown); Donor(1, thrown); TaomEquipmentCompatibility.Reset();
        EquipHero.Adopt(archer, new Settings(), archer.Class);
        Check(archer.BattleEquipment[EquipmentIndex.Weapon0].Item == thrown, "Ranged fallback can use throwing weapons without separate ammo");
        archer.Class.SlotItems = new[] { EquipmentType.Shield, EquipmentType.None };
        EquipHero.Adopt(archer, new Settings(), archer.Class);
        Check(!archer.BattleEquipment.YieldFilledWeaponSlots().Any(), "Shield and empty class slots do not become weapons");
        Reset(); var dwarf = Item("dwarf", 0); var horse = Item("human-horse", 0, ItemObject.ItemTypeEnum.Horse); horse.HorseComponent = new(); Donor(0, horse); Donor(3, dwarf);
        var rider = Hero(3); rider.Class.Mounted = rider.Class.UseHorse = true;
        EquipHero.UpgradeEquipment(rider, 0, rider.Class, true);
        Check(rider.BattleEquipment[EquipmentIndex.Horse].IsEmpty, "Dwarf does not borrow a mount from a different race");
        var customHero = Hero(3, human); customHero.CustomItems.Add(new(human) { ItemModifier = new() });
        EquipHero.UpgradeEquipment(customHero, 0, customHero.Class, false);
        Check(customHero.BattleEquipment[EquipmentIndex.Body].Item == dwarf, "Custom-item and retained-gear paths cannot bypass race compatibility");
        Check(!TaomEquipmentCompatibility.CanUse(Hero(99), human), "Unknown races fail closed");
        var live = Hero(3, human); live.CharacterObject.BattleEquipments.Add(live.BattleEquipment); CharacterObject.All.Add(live.CharacterObject); TaomEquipmentCompatibility.Reset();
        Check(!TaomEquipmentCompatibility.CanUse(live, human), "Live hero gear cannot contaminate the compatibility catalogue");
        BLTAdoptAHeroModule.CommonConfig.RestrictedItemIds.Add("dwarf");
        Check(EquipHero.FindRandomTieredEquipment(0, Hero(3), false) == null, "Restricted items remain excluded");
        Campaign.Current = new(); CharacterObject.All.Clear();
        Check(!TaomEquipmentCompatibility.CanUse(Hero(3), dwarf), "Campaign change does not reuse the old catalogue");
        Reset(); HarmonyLib.AccessTools.Installed = false; CampaignHelpers.AllItems.Add(human);
        Check(EquipHero.FindRandomTieredEquipment(0, Hero(3), false) == human, "Without TAOM the ordinary item pool still works");
        foreach (var (slot, type) in SkillGroup.ArmorIndexType)
        {
            Reset();
            var ownCulture = new CultureObject(); var foreignCulture = new CultureObject();
            var own = Item("own-armour", 1, type); own.Culture = ownCulture;
            var foreignArmour = Item("foreign-armour", 0, type); foreignArmour.Culture = foreignCulture;
            Donor(1, own, foreignArmour);
            var wearer = Hero(1); wearer.Culture = ownCulture;
            EquipHero.Adopt(wearer, new Settings(), wearer.Class);
            Check(wearer.BattleEquipment[slot].Item == own && wearer.CivilianEquipment[slot].Item == own && wearer.Tier == 0,
                $"{slot}: same-culture adjacent-tier armour preferred without tier inflation");
            own.Tier = 5;
            EquipHero.Adopt(wearer, new Settings(), wearer.Class);
            Check(wearer.BattleEquipment[slot].Item == foreignArmour && wearer.CivilianEquipment[slot].Item == foreignArmour,
                $"{slot}: any compatible foreign armour fills battle and civilian slots");
            foreignArmour.IsCivilian = false;
            EquipHero.Adopt(wearer, new Settings(), wearer.Class);
            Check(wearer.BattleEquipment[slot].Item == foreignArmour && wearer.CivilianEquipment[slot].IsEmpty,
                $"{slot}: civilian restrictions survive fallback");
            CampaignHelpers.AllItems.Remove(foreignArmour);
            var trollArmour = Item("troll-armour", 0, type); Donor(8, trollArmour);
            var wrongSlot = Item("wrong-slot", 0, type == ItemObject.ItemTypeEnum.BodyArmor ? ItemObject.ItemTypeEnum.HeadArmor : ItemObject.ItemTypeEnum.BodyArmor);
            Donor(1, wrongSlot); TaomEquipmentCompatibility.Reset();
            EquipHero.Adopt(wearer, new Settings(), wearer.Class);
            Check(wearer.BattleEquipment[slot].IsEmpty, $"{slot}: fallback rejects wrong race, wrong slot and distant tiers");
        }
        Reset();
        var gondor = new CultureObject(); var foreignCulture2 = new CultureObject();
        var fallbackMace = Item("ordinary-mace", 0, ItemObject.ItemTypeEnum.OneHandedWeapon); fallbackMace.Culture = foreignCulture2;
        var fallbackShield = Item("ordinary-shield", 0, ItemObject.ItemTypeEnum.Shield);
        var fallbackAxes = Item("ordinary-axes", 0, ItemObject.ItemTypeEnum.Thrown);
        // None of these items appears in ANY troop template: missing evidence must not forbid weapons.
        CampaignHelpers.AllItems.AddRange(new[] { fallbackMace, fallbackShield, fallbackAxes });
        var gondorian = Hero(1); gondorian.Culture = gondor;
        gondorian.Class.Name = "Dwarf Warrior";
        gondorian.Class.SlotItems = new[] { EquipmentType.OneHandedMace, EquipmentType.Shield, EquipmentType.ThrowingAxes, EquipmentType.ThrowingAxes };
        EquipHero.Adopt(gondorian, new Settings(), gondorian.Class);
        Check(gondorian.BattleEquipment.YieldFilledWeaponSlots().Count() == 4 && gondorian.Tier == 0,
            "Gondor Dwarf Warrior gets mace, shield and throwing axes without same-race template evidence");
        Check(gondorian.BattleEquipment[EquipmentIndex.Weapon0].Item == fallbackMace,
            "Foreign culture ordinary weapon survives race filtering");
        foreach (var marker in new[] { "wm_cave_troll_1h_mace_a", "olog_hammer", "giant_club", "ent_branch", "balrog_sword" })
            Check(!EquipHero.CanUseItem(gondorian, Item(marker, 0, ItemObject.ItemTypeEnum.OneHandedWeapon), true, false),
                "Creature-marked weapon remains protected: " + marker);
        var ownTrollClub = Item("troll_club", 0, ItemObject.ItemTypeEnum.OneHandedWeapon);
        Donor(8, ownTrollClub); TaomEquipmentCompatibility.Reset();
        Check(EquipHero.CanUseItem(Hero(8), ownTrollClub, true, false), "Creature may still use its own authored weapon");
        fallbackMace.RelevantSkill = new SkillObject(); fallbackMace.Difficulty = 101;
        Check(!EquipHero.CanUseItem(gondorian, fallbackMace, false, false), "Shared weapon still checks skill requirements");
        fallbackMace.Difficulty = 0; BLTAdoptAHeroModule.CommonConfig.RestrictedItemIds.Add(fallbackMace.StringId);
        EquipHero.Adopt(gondorian, new Settings(), gondorian.Class);
        Check(gondorian.BattleEquipment[EquipmentIndex.Weapon0].IsEmpty, "Shared ordinary weapon still respects restricted item IDs");
        Console.WriteLine($"{checks} checks passed.");
    }
