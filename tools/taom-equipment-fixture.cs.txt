public static class Program
{
    static int checks;
    static void Check(bool value, string description) { if (!value) throw new Exception(description); Console.WriteLine("PASS " + description); checks++; }
    static ItemObject Item(string id, int tier, ItemObject.ItemTypeEnum type = ItemObject.ItemTypeEnum.BodyArmor) => new() { StringId = id, Tier = tier, ItemType = type };
    static void Reset()
    {
        Campaign.Current = new(); CharacterObject.All.Clear(); CampaignHelpers.AllItems.Clear();
        HarmonyLib.AccessTools.Installed = true; TaomEquipmentCompatibility.Reset();
        BLTAdoptAHeroModule.CommonConfig.RestrictedItemIds.Clear();
    }
    static void Donor(int race, params ItemObject[] items)
    {
        var c = new CharacterObject { Race = race };
        foreach (var item in items) { var e = new Equipment(); e[EquipmentIndex.Body] = new(item); c.BattleEquipments.Add(e); }
        CharacterObject.All.Add(c); CampaignHelpers.AllItems.AddRange(items);
    }
    static Hero Hero(int race, ItemObject initial = null)
    {
        var h = new Hero { CharacterObject = new() { Race = race, IsHero = true } };
        h.BattleEquipment[EquipmentIndex.Body] = new(initial); h.CivilianEquipment[EquipmentIndex.Body] = new(initial); return h;
    }
    public static void Main()
    {
        Reset();
        var human = Item("human", 0); var east = Item("east", 0); var elf = Item("elf", 0); var elite = Item("elf-elite", 5);
        var sword = Item("elf-sword", 0, ItemObject.ItemTypeEnum.OneHandedWeapon);
        Donor(0, human); Donor(1, east); Donor(2, elf, elite, sword);
        Check(!EquipHero.CanUseItem(Hero(2), human, true, false), "Elf excludes human gear even when ability is ignored");
        Check(!EquipHero.CanUseItem(Hero(1), elf, true, false), "Custom human race does not receive elf armour");
        foreach (var race in new[] { 1, 2 })
        {
            var h = Hero(race, elite); h.CustomItems.Add(new(elite));
            EquipHero.Adopt(h, new Settings(), h.Class);
            Check(h.Tier == 0, $"Race {race} starts at displayed Tier 1");
            Check(h.BattleEquipment.YieldFilledEquipmentSlots().All(x => x.element.Item.Tier <= 0), $"Race {race} has no template/inherited Tier 6 battle gear");
            Check(h.CivilianEquipment.YieldFilledEquipmentSlots().All(x => x.element.Item.Tier <= 0), $"Race {race} has no Tier 6 civilian gear");
            Check(h.BattleEquipment[EquipmentIndex.Body].Item == (race == 1 ? east : elf), $"Race {race} receives its own compatible armour");
        }
        var noGear = Hero(2, elite); EquipHero.Adopt(noGear, new Settings { StartingEquipmentTier = 0 }, noGear.Class);
        Check(noGear.Tier == -1 && !noGear.BattleEquipment.YieldFilledEquipmentSlots().Any() && !noGear.CivilianEquipment.YieldFilledEquipmentSlots().Any(), "Starting tier zero clears both loadouts for custom races");
        Reset(); Donor(2, elite);
        var missing = Hero(2, elite); EquipHero.Adopt(missing, new Settings(), missing.Class);
        Check(!missing.BattleEquipment.YieldFilledEquipmentSlots().Any(), "No low-tier kit means an empty slot, never elite fallback");
        var changed = true; EquipHero.Paid(missing, new Settings(), _ => changed = true, _ => changed = false);
        Check(!changed && missing.Gold == 1000 && missing.Tier == 0, "No suitable equipment does not charge or advance tier");
        Reset(); var culture = new CultureObject(); elf.Culture = culture; elite.Culture = culture; Donor(2, elf, elite);
        Check(EquipHero.FindRandomTieredEquipment(0, Hero(2), false, cultureFilter: culture, cultureFilterSpecified: true) == elf, "Culture selection obeys tier instead of taking the best item");
        var foreign = new CultureObject(); human.Culture = foreign; Donor(0, human); TaomEquipmentCompatibility.Reset();
        var fallback = Hero(2); EquipHero.UpgradeEquipment(fallback, 0, fallback.Class, true, foreign, true);
        Check(fallback.BattleEquipment[EquipmentIndex.Body].Item == elf, "Culture fallback retains race and tier checks");
        var upgraded = Item("elf-tier2", 1); Donor(2, upgraded); TaomEquipmentCompatibility.Reset();
        EquipHero.Paid(fallback, new Settings(), _ => changed = true, _ => changed = false);
        Check(changed && fallback.BattleEquipment[EquipmentIndex.Body].Item == upgraded && fallback.Gold == 900 && fallback.Tier == 1, "Non-human upgrade applies gear and charges once");
        var alternative = Item("elf-tier2-alternate", 1); Donor(2, alternative); TaomEquipmentCompatibility.Reset();
        EquipHero.Paid(fallback, new Settings { ReequipInsteadOfUpgrade = true }, _ => changed = true, _ => changed = false);
        Check(changed && fallback.BattleEquipment[EquipmentIndex.Body].Item == alternative && fallback.Tier == 1, "Non-human re-equip selects an alternative at the same tier");
        Reset(); var dwarf = Item("dwarf", 0); var horse = Item("human-horse", 0, ItemObject.ItemTypeEnum.Horse); horse.HorseComponent = new(); Donor(0, horse); Donor(3, dwarf);
        var rider = Hero(3); rider.Class.Mounted = rider.Class.UseHorse = true;
        EquipHero.UpgradeEquipment(rider, 0, rider.Class, true);
        Check(rider.BattleEquipment[EquipmentIndex.Horse].IsEmpty, "Dwarf does not borrow a mount from a different race");
        var customHero = Hero(3, human); customHero.CustomItems.Add(new(human) { ItemModifier = new() });
        EquipHero.UpgradeEquipment(customHero, 0, customHero.Class, false);
        Check(customHero.BattleEquipment[EquipmentIndex.Body].Item == dwarf, "Custom-item and retained-gear paths cannot bypass race compatibility");
        Check(!TaomEquipmentCompatibility.CanUse(Hero(99), human), "Unknown races fail closed");
        var live = Hero(3, human); live.CharacterObject.BattleEquipments.Add(live.BattleEquipment); CharacterObject.All.Add(live.CharacterObject); TaomEquipmentCompatibility.Reset();
        Check(!TaomEquipmentCompatibility.CanUse(live, human), "Live hero gear cannot contaminate the compatibility catalogue");
        BLTAdoptAHeroModule.CommonConfig.RestrictedItemIds.Add("dwarf");
        Check(EquipHero.FindRandomTieredEquipment(0, Hero(3), false) == null, "Restricted items remain excluded");
        Campaign.Current = new(); CharacterObject.All.Clear();
        Check(!TaomEquipmentCompatibility.CanUse(Hero(3), dwarf), "Campaign change does not reuse the old catalogue");
        Reset(); HarmonyLib.AccessTools.Installed = false; CampaignHelpers.AllItems.Add(human);
        Check(EquipHero.FindRandomTieredEquipment(0, Hero(3), false) == human, "Without TAOM the ordinary item pool still works");
        Console.WriteLine($"{checks} checks passed.");
    }
